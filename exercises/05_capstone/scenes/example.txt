# Example scene for the ray tracer.
# Lines starting with '#' are comments. Blank lines are ignored.

# Image resolution in pixels.
size 800 600

# Camera:  eye position | look-at point | up vector | vertical field of view
camera  0 1.5 5    0 0.5 0    0 1 0    50

# Colours for rays that miss everything, and for uniform ambient light.
background 0.5 0.7 1.0
ambient    0.15 0.15 0.15

# Point lights:  position | colour
light  5  5  5    1.0 1.0 1.0
light -3  4  2    0.3 0.3 0.3

# Materials:  name | diffuse rgb | specular rgb | shininess | reflectivity
material red    0.8 0.2 0.2   1.0 1.0 1.0   64   0.1
material blue   0.2 0.3 0.9   1.0 1.0 1.0   64   0.1
material mirror 0.1 0.1 0.1   1.0 1.0 1.0  256   0.8
material floor  0.6 0.6 0.6   0.2 0.2 0.2   16   0.2

# Three spheres sitting on the floor:  center | radius | material
sphere -1.2 0.5  0.0   0.5  red
sphere  0.0 0.5  0.3   0.5  blue
sphere  1.3 0.7 -0.6   0.7  mirror

# The floor, made of two triangles forming a big quad at y = 0.
#   v0 | v1 | v2 | material
triangle -4 0 -4    4 0 -4    4 0 4    floor
triangle -4 0 -4    4 0  4   -4 0 4    floor
