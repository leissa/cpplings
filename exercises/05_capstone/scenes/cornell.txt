# A Cornell-box style room built entirely from triangles.
# Red left wall, green right wall, white floor/ceiling/back wall,
# two spheres inside (one matte, one mirror).

size 600 600

camera  0 2 9    0 2 -1    0 1 0    45

background 0.0 0.0 0.0
