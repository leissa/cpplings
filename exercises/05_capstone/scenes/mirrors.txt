# Two facing mirror walls with a sphere between them.
# The recursive reflection produces a hall-of-mirrors effect, cut off after
# the renderer's maximum recursion depth.

size 700 600

camera  0 1.5 7    0 1 0    0 1 0    50

background 0.7 0.8 0.95
ambient    0.15 0.15 0.15

light  0 5 3     1.0 1.0 1.0

material floor  0.6 0.6 0.6   0.2 0.2 0.2   16   0.1
material mirror 0.05 0.05 0.05  1.0 1.0 1.0  256  0.9
material ball   0.9 0.4 0.1   1.0 1.0 1.0   64   0.2

# Floor (y = 0)
triangle -6 0 -6    6 0 -6    6 0 6   floor
triangle -6 0 -6    6 0  6   -6 0 6   floor

# Left mirror wall (x = -2.5)
triangle -2.5 0 -3   -2.5 0 3   -2.5 4 3    mirror
triangle -2.5 0 -3   -2.5 4 3   -2.5 4 -3   mirror

# Right mirror wall (x = 2.5)
triangle 2.5 0 -3    2.5 0 3    2.5 4 3     mirror
triangle 2.5 0 -3    2.5 4 3    2.5 4 -3    mirror

# The sphere caught between the mirrors.
sphere 0 1 0   1.0  ball
